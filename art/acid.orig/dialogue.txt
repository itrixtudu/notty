static inline void fire_color(int iters, int max) {

// Mandelbrot formula
static int mandelbrot(double x0, double y0, int maxIter) {
    double x = 0, y = 0;
    int iter = 0;
    while (x*x + y*y <= 4 && iter < maxIter) {
        double xt = x*x - y*y + x0;
        y = 2*x*y + y0;
        x = xt;
        iter++;
    }
    return iter;
}

// Julia formula
static int julia(double x, double y, int maxIter, double cx, double cy) {
    int iter = 0;
    while (x*x + y*y <= 4 && iter < maxIter) {
        double xt = x*x - y*y + cx;
        y = 2*x*y + cy;
        x = xt;
        iter++;
    }
    return iter;
}

static void draw_fractal(int useJulia) {
    printf("\033[2J\033[H");

    int maxIter = 80;
    int width = 90;
    int height = 40;

    double cx = -0.7;   // Julia constants
    double cy = 0.27015;

    for (int py = 0; py < height; py++) {
        double y = (py - height/2.0) * 0.035;

        for (int px = 0; px < width; px++) {
            double x = (px - width/2.0) * 0.035;

            int it = useJulia
                ? julia(x, y, maxIter, cx, cy)
                : mandelbrot(x, y, maxIter);

            fire_color(it, maxIter);

            putchar(it == maxIter ? ' ' : '#');
        }
        printf("\033[0m\n");
    }

    printf("\n\033[37m[ press ENTER to unlock | ESC to cancel ]\033[0m\n");
}

int fire_palette(int it) {
    // it: iteration count [0, MAX]
    // clamp to range 0–100 for smoother fire
    if (it > 100) it = 100;

    // Map to warm tones.
    // 0 = darkest red
    // 100 = bright white
    int p = it;

    if (p < 20)  return 196; // deep red
    if (p < 40)  return 202; // red-orange
    if (p < 60)  return 208; // orange
    if (p < 80)  return 214; // yellow-orange
    if (p < 95)  return 226; // yellow
    return 231;              // white-hot
}

int julia(double x, double y, double cx, double cy) {

        double xt = x*x - y*y + cx;
        y = 2*x*y + cy;

void draw_frame(double cx, double cy, double zoom) {
    printf("\033[H"); // cursor home

    for (int j = 0; j < H; j++) {

            double x0 = (i - W / 2) * zoom;

// ──────────────────────────────────────────────────────────
//   FRACTAL GENERATION (MANDELBROT & JULIA)
// ──────────────────────────────────────────────────────────

// Mandelbrot formula
static int mandelbrot(double x0, double y0, int maxIter) {
    double x = 0, y = 0;
    int iter = 0;
    while (x*x + y*y <= 4 && iter < maxIter) {
        double xt = x*x - y*y + x0;
        y = 2*x*y + y0;
        x = xt;
        iter++;
    }
    return iter;
}

// Julia formula
static int julia(double x, double y, int maxIter, double cx, double cy) {
    int iter = 0;
    while (x*x + y*y <= 4 && iter < maxIter) {
        double xt = x*x - y*y + cx;
        y = 2*x*y + cy;
        x = xt;
        iter++;
    }
    return iter;
}

// ──────────────────────────────────────────────────────────
//   FRACTAL RENDERING
// ──────────────────────────────────────────────────────────
//

static void draw_fractal(int useJulia) {
    printf("\033[2J\033[H");

    int maxIter = 80;
    int width = 90;
    int height = 40;

    double cx = -0.7;   // Julia constants
    double cy = 0.27015;

    for (int py = 0; py < height; py++) {
        double y = (py - height/2.0) * 0.035;

        for (int px = 0; px < width; px++) {
            double x = (px - width/2.0) * 0.035;

            int it = useJulia
                ? julia(x, y, maxIter, cx, cy)
                : mandelbrot(x, y, maxIter);

            fire_color(it, maxIter);

            putchar(it == maxIter ? ' ' : '#');
        }
        printf("\033[0m\n");
    }

    printf("\n\033[37m[ press ENTER to unlock | ESC to cancel ]\033[0m\n");
}

    const char *file = NULL;

    // Argument parsing
    for (int i = 1; i < argc; i++) {
        if (!strcmp(argv[i], "--help")) {
            show_help(argv[0]);
            return 0;
        }
        if (!strcmp(argv[i], "--mandelbrot")) {
            useFractal = 1;
            useJulia = 0;
        }
        else if (!strcmp(argv[i], "--julia")) {
            useFractal = 1;
            useJulia = 1;
        }
        else {
            file = argv[i];
        }
    }

    const char *unlock = getenv("UNLOCK_CODE");
    if (!unlock) unlock = "";

    raw_mode();
    char input[256];

    char *art = NULL;
    if (file && !useFractal)
        art = load_file(file);

    for (;;) {
        memset(input, 0, sizeof(input));
        int pos = 0;
        time_t start = time(NULL);

        for (;;) {
            if (useFractal)
                draw_fractal(useJulia);
            else
                draw_art(art, pos);

                        return 0;
                    }
                    break;
                }
                if (pos < 255) input[pos++] = c;
            }

            if (time(NULL) - start > 12)

        }
    }

    free(art);
    return 0;
}

// Returns the number of bytes in the UTF-8 codepoint starting at *s
static int utf8_len(const unsigned char *s) {
    if (*s < 0x80) return 1;
    if ((*s >> 5) == 0x6) return 2;
    if ((*s >> 4) == 0xE) return 3;
    if ((*s >> 3) == 0x1E) return 4;
    return 1; // fallback
}

// count characters (grapheme count)
static int utf8_strlen(const char *s) {
    int count = 0;
    while (*s) {
        s += utf8_len((unsigned char*)s);
        count++;
    }
    return count;
}

// copy a single UTF-8 grapheme into buf
static int utf8_copy_char(char *buf, const char *src) {
    int n = utf8_len((unsigned char*)src);
    for (int i = 0; i < n; i++)
        buf[i] = src[i];
    buf[n] = 0;
    return n;
}

// ─────────────────────────────────────────────────────────────
//   Terminal control
// ─────────────────────────────────────────────────────────────

static struct termios old_tio;

static void restore_tio() {
    tcsetattr(STDIN_FILENO, TCSANOW, &old_tio);
}

static void raw_mode() {
    struct termios new_tio;
    tcgetattr(STDIN_FILENO, &old_tio);
    atexit(restore_tio);

    new_tio = old_tio;
    new_tio.c_lflag &= ~(ICANON | ECHO);
    new_tio.c_cc[VMIN] = 0;
    new_tio.c_cc[VTIME] = 0;

    tcsetattr(STDIN_FILENO, TCSANOW, &new_tio);
}

// get terminal size
static void get_terminal_size(int *w, int *h) {
    struct winsize ws;
    if (ioctl(STDOUT_FILENO, TIOCGWINSZ, &ws) == 0) {
        *w = ws.ws_col;
        *h = ws.ws_row;
    } else {
        *w = 120;
        *h = 40;
    }
}

                if (!strcmp(argv[i], "overlay")) cfg.banner_mode = 1;
                else if (!strcmp(argv[i], "mask")) cfg.banner_mode = 2;
                else if (!strcmp(argv[i], "cutout")) cfg.banner_mode = 3;

            if (i+1 < argc) {
                i++;
                if (!strcmp(argv[i], "ascii")) cfg.glyph_mode = 0;
                else if (!strcmp(argv[i], "blocks")) cfg.glyph_mode = 1;
                else if (!strcmp(argv[i], "braille")) cfg.glyph_mode = 2;

    }
